# Имена боевого окружения, которые читают скрипты хоста (бэкап, мониторинг), а не контейнеры:
# отсутствие в compose v2 для них не потеря настройки (регулярные выражения Python).
^BACKUP_
^(DATA_DIR|SECRETS_DIR|MTLS_DIR|MODELS_DIR|TEXTFILE_METRICS_DIR)$
^(SMB_|MOUNT_)
_HOST_DIR$
_HOST_FILE$
^COMPOSE_
# Сроки хранения и имя тома приёма читают maintenance.sh и backup-lib.sh старого дерева на хосте.
_RETENTION_DAYS$
^INTAKE_VOLUME$
# С платформы 1.0.1 образы postgres, nginx, docker-proxy и alloy берутся только из images.lock;
# старые ключи боевого .env базой не читаются (bai их снимает и предупреждает), это не потеря
# настройки.
^(POSTGRES_IMAGE|NGINX_IMAGE|DOCKER_PROXY_IMAGE|ALLOY_IMAGE)$
